cmake_minimum_required(VERSION 3.16)

# The version comes from the top-level project() so the app, the macOS
# bundle and the installer all agree on it.
project(WisdomChessQtQml VERSION ${WisdomChess_VERSION} LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Qt 6.8+ is required for the QTP0004 policy below (and the QML files
# rely on its generated subdirectory qmldir files for implicit imports).
find_package(Qt6 6.8 COMPONENTS Quick Svg REQUIRED)
qt_policy(SET QTP0001 NEW)
qt_policy(SET QTP0004 NEW)

if(APPLE AND TARGET WrapOpenGL::WrapOpenGL)
    # Qt < 6.8.4 / 6.9.2 links the AGL framework via its FindWrapOpenGL
    # module, but Apple removed AGL from the macOS 26 SDK (QTBUG-137687),
    # which breaks linking with "ld: framework 'AGL' not found". Nothing
    # uses AGL symbols (it is dead Carbon-era glue), so strip it
    # unconditionally whether Qt added it as a flag or as a found path.
    get_target_property(_wisdom_wrap_opengl_libs WrapOpenGL::WrapOpenGL
        INTERFACE_LINK_LIBRARIES)
    if(_wisdom_wrap_opengl_libs)
        list(FILTER _wisdom_wrap_opengl_libs
            EXCLUDE REGEX "-framework AGL|AGL\\.framework")
        set_target_properties(WrapOpenGL::WrapOpenGL PROPERTIES
            INTERFACE_LINK_LIBRARIES "${_wisdom_wrap_opengl_libs}")
    endif()
endif()

if(EMSCRIPTEN)
    # Qt's wasm_multithread uses pthreads, which combined with ALLOW_MEMORY_GROWTH
    # can slow down JavaScript glue code. This is acceptable for a chess engine where
    # most computation happens in WASM. Suppress the warning but inform at configure time.
    # See: https://github.com/WebAssembly/design/issues/1271
    message(STATUS "wisdom-chess: QML WASM build uses pthreads + memory growth (may affect JS glue performance)")
    target_link_options(WisdomChessQml PRIVATE
        -Wno-pthreads-mem-growth
    )

    # Verify Emscripten version matches Qt requirements
    # Qt prebuilt binaries require specific Emscripten versions for ABI compatibility
    if(DEFINED QT_VERSION_MAJOR AND DEFINED QT_VERSION_MINOR)
        if(QT_VERSION_MAJOR EQUAL 6 AND QT_VERSION_MINOR EQUAL 6)
            set(EXPECTED_EMSCRIPTEN "3.1.37")
        elseif(QT_VERSION_MAJOR EQUAL 6 AND QT_VERSION_MINOR EQUAL 7)
            set(EXPECTED_EMSCRIPTEN "3.1.50")
        elseif(QT_VERSION_MAJOR EQUAL 6 AND QT_VERSION_MINOR EQUAL 8)
            set(EXPECTED_EMSCRIPTEN "3.1.56")
        elseif(QT_VERSION_MAJOR EQUAL 6 AND QT_VERSION_MINOR EQUAL 9)
            set(EXPECTED_EMSCRIPTEN "3.1.70")
        elseif(QT_VERSION_MAJOR EQUAL 6 AND QT_VERSION_MINOR EQUAL 10)
            set(EXPECTED_EMSCRIPTEN "4.0.7")
        elseif(QT_VERSION_MAJOR EQUAL 6 AND QT_VERSION_MINOR EQUAL 11)
            set(EXPECTED_EMSCRIPTEN "4.0.7")
        endif()

        if(DEFINED EXPECTED_EMSCRIPTEN AND NOT EMSCRIPTEN_VERSION VERSION_EQUAL EXPECTED_EMSCRIPTEN)
            message(FATAL_ERROR "Qt ${QT_VERSION} requires Emscripten ${EXPECTED_EMSCRIPTEN}, "
                               "but you are using ${EMSCRIPTEN_VERSION}. "
                               "Please install the correct version: ./emsdk install ${EXPECTED_EMSCRIPTEN} && ./emsdk activate ${EXPECTED_EMSCRIPTEN}")
        endif()
    endif()

    target_compile_definitions(WisdomChessQml PRIVATE USE_DEFAULT_GRAPHICS_BACKEND)
elseif(APPLE)
    # OpenGL is deprecated on macOS (and AGL was removed in macOS 26), so
    # use Qt's default scene graph backend there (Metal).
    target_compile_definitions(WisdomChessQml PRIVATE USE_DEFAULT_GRAPHICS_BACKEND)
else()
    target_compile_definitions(WisdomChessQml PRIVATE USE_OPENGL_GRAPHICS_BACKEND)
endif()

include(GNUInstallDirs)

if(UNIX AND NOT APPLE)
    # Qt's deploy step puts the Qt libraries in <prefix>/lib and only rewrites
    # the RPATH of the plugins it copies, so the executable has to find
    # ../lib on its own once installed.
    set_target_properties(WisdomChessQml PROPERTIES
        INSTALL_RPATH "$ORIGIN/../${CMAKE_INSTALL_LIBDIR}")
endif()

# BUNDLE DESTINATION must stay relative: Qt's bundle deployment assumes the
# .app sits at the top of the install prefix, and CPack stages installs
# under its own prefix when building the installer.
# COMPONENT only applies to the artifact group it follows, so it has to be
# repeated for the bundle (macOS) and the plain executable (Linux, Windows).
install(TARGETS WisdomChessQml
    BUNDLE DESTINATION . COMPONENT Application
    RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}" COMPONENT Application)

if(WISDOM_CHESS_INSTALLER)
    # Deploy the Qt runtime (libraries, plugins, QML modules, qt.conf) next
    # to the installed executable. This has to be generated from this
    # directory: QT_DEPLOY_SUPPORT is scoped to the directory that called
    # find_package(Qt6).
    set(wisdom_deploy_args NO_TRANSLATIONS)
    if(WIN32)
        # vc_redist.x64.exe is bundled and run by the installer instead of
        # letting windeployqt drop loose CRT DLLs next to the executable.
        list(APPEND wisdom_deploy_args NO_COMPILER_RUNTIME)
    endif()
    if(UNIX AND NOT APPLE AND Qt6_VERSION VERSION_GREATER_EQUAL 6.10)
        # The plugin selection keywords only exist since Qt 6.10 (unknown
        # keywords are a fatal error). The Wayland client plugin types are
        # useless without the qwayland platform plugin, which Qt's deploy
        # step does not ship by default.
        list(APPEND wisdom_deploy_args EXCLUDE_PLUGIN_TYPES
            qmltooling egldeviceintegrations
            wayland-graphics-integration-client wayland-decoration-client
            wayland-shell-integration)
    endif()
    qt_generate_deploy_qml_app_script(
        TARGET WisdomChessQml
        OUTPUT_SCRIPT wisdom_deploy_script
        ${wisdom_deploy_args})
    install(SCRIPT "${wisdom_deploy_script}" COMPONENT Application)

    if(UNIX AND NOT APPLE)
        # Referenced by the .desktop entry the installer creates.
        install(FILES "${CMAKE_SOURCE_DIR}/installer/icons/wisdom-chess-256.png"
            DESTINATION share/icons RENAME wisdom-chess.png
            COMPONENT Application)
    endif()
    if(WIN32)
        install(FILES "${WISDOM_CHESS_VCREDIST}"
            DESTINATION "${CMAKE_INSTALL_BINDIR}" COMPONENT Application)
    endif()
endif()
